<script>

    // 提示框

    function showToast(msg) {

        const toast = document.getElementById('toast');

        toast.textContent = msg;

        toast.classList.add('show');

        setTimeout(() => toast.classList.remove('show'), 2000);

    }

 

    // 清除所有勾选

    function clearAll() {

        const inputs = document.querySelectorAll('input[type="checkbox"], input[type="radio"]');

        inputs.forEach(input => input.checked = false);

        showToast('已重置所有选择');

    }

 

    // 生成订单

    function generateOrder() {

        let order = "🍽️ 我的健康早餐选择：\n\n";

       

        // 获取单选框（蛋白质1）

        const protein1 = document.querySelector('input[name="protein1"]:checked');

        if (protein1) order += `✅ 蛋白质：${protein1.value}\n`;

 

        // 获取多选框（燕麦）

        const oatmeal = document.getElementById('oatmeal');

        if (oatmeal.checked) order += `✅ 蛋白质：${oatmeal.value}\n`;

 

        // 获取单选框（蔬菜）

        const veg = document.querySelector('input[name="veg"]:checked');

        if (veg) order += `✅ 蔬菜类：${veg.value}\n`;

 

        // 获取多选框（碳水）

        const carbs = [];

        document.querySelectorAll('#carb1, #carb2, #carb3, #carb4, #carb5').forEach(cb => {

            if (cb.checked) carbs.push(cb.value);

        });

        if (carbs.length > 0) order += `✅ 碳水类：${carbs.join('、')}\n`;

 

        // 获取多选框（脂肪）

        const fat1 = document.getElementById('fat1');

        if (fat1.checked) order += `✅ 脂肪类：${fat1.value}\n`;

 

        if (order === "🍽️ 我的健康早餐选择：\n\n") {

            showToast('请先勾选您想吃的食物哦！');

            return;

        }

 

        order += "\n📝 请家人根据此菜单准备，谢谢！";

       

        document.getElementById('order-text').value = order;

        document.getElementById('result-modal').style.display = 'block';

    }

 

    // 复制文字

    function copyOrder() {

        const textarea = document.getElementById('order-text');

        textarea.select();

        document.execCommand('copy');

        showToast('复制成功！请粘贴发给家人吧');

    }

 

    // 关闭弹窗

    function closeModal() {

        document.getElementById('result-modal').style.display = 'none';

    }

</script>
